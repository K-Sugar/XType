## Step E2.1 — Latency tracking in the engine

**What:** Record wall-clock time from when `requestInference()` fires to
when the `on_done` callback fires. Maintain a rolling window of the last
32 samples to compute P50 and P95. Write the result to
`~/.local/share/xtype/metrics.json` after each inference, gated at
≤ 1 write/sec.

**File:** `fcitx5-engine/src/xtype.h`

Add to the private section:

```cpp
    // Latency tracking: rolling window of last 32 inference durations (ms).
    static constexpr size_t kLatencyWindowSize = 32;
    std::vector<int>         _latencyWindow;
    std::time_t              _metricsLastWrite{0};  // rate-limit metrics.json

    void recordLatency(int ms);          // updates _latencyWindow + _metrics
    void writeMetrics();                 // writes ~/.local/share/xtype/metrics.json
```

**File:** `fcitx5-engine/src/xtype.cpp`

In `requestInference()`, capture start time before calling `_inference.request()`:

```cpp

    fcitx::TrackableObjectReference<fcitx::InputContext> icRef,
    fcitx::InputContext *icPtr)
{
    // ... (existing context-size checks) ...
    ++_gen;
    ++_metrics.suggestions_generated;
    const uint64_t myGen = _gen;
    const auto     startUs = fcitx::now(CLOCK_MONOTONIC);   // <-- ADD

    _inference.request(
        std::move(ctx),
        /* on_token */ [this, myGen, icRef, icPtr](std::string token) {
            // ... unchanged ...
        },
        /* on_done */ [this, myGen, icRef, icPtr, startUs]() {  // <-- capture startUs

}
```

Implement `recordLatency()` and `writeMetrics()` in `xtype.cpp`:

```cpp
void XTypeEngine::recordLatency(int ms) {
    if (_latencyWindow.size() >= kLatencyWindowSize)
        _latencyWindow.erase(_latencyWindow.begin());
    _latencyWindow.push_back(ms);

    // Compute P50 and P95 from sorted copy.
    std::vector<int> sorted = _latencyWindow;
    std::sort(sorted.begin(), sorted.end());
    size_t n = sorted.size();
    _metrics.latency_p50_ms.store(sorted[n / 2]);
    _metrics.latency_p95_ms.store(sorted[std::min(n - 1, (size_t)(n * 0.95))]);

    writeMetrics();
}

void XTypeEngine::writeMetrics() {
    // Rate-limit to 1 write per second.
    std::time_t now = std::time(nullptr);
    if (now == _metricsLastWrite) return;
    _metricsLastWrite = now;

    auto dir = path_utils::expandTilde("~/.local/share/xtype");
    auto tmp = (dir / "metrics.json.tmp").string();
    auto dst = (dir / "metrics.json").string();

    char buf[512];
    std::snprintf(buf, sizeof(buf),
        "{\"latency_p50_ms\":%d,\"latency_p95_ms\":%d,"
        "\"suggestions_generated\":%" PRIu64 ","
        "\"suggestions_accepted\":%" PRIu64 ","
        "\"chars_accepted\":%" PRIu64 ","
        "\"updated_at\":%" PRId64 "}\n",
        _metrics.latency_p50_ms.load(),
        _metrics.latency_p95_ms.load(),
        (uint64_t)_metrics.suggestions_generated.load(),
        (uint64_t)_metrics.suggestions_accepted.load(),
        (uint64_t)_metrics.chars_accepted.load(),
        (int64_t)now);

    // Atomic write: write to .tmp then rename.
    if (FILE* f = std::fopen(tmp.c_str(), "w")) {
        std::fputs(buf, f);
        std::fclose(f);
        std::rename(tmp.c_str(), dst.c_str());
    }
}
```

Add `#include <cinttypes>` to `xtype.cpp` for `PRIu64`/`PRId64`.

```cpp
void XTypeEngine::writeRecentEvents() {
    std::time_t now = std::time(nullptr);
    if (now == _eventsLastWrite) return;
    _eventsLastWrite = now;

    auto dir = path_utils::expandTilde("~/.local/share/xtype");
    auto tmp = (dir / "recent_events.json.tmp").string();
    auto dst = (dir / "recent_events.json").string();

    auto snap = _recent.snapshot();     // newest-first
    std::string json = "[\n";
    bool first = true;
    for (const auto& ev : snap) {
        if (!first) json += ",\n";
        first = false;
        char entry[1024];
        // json_escape inline via a lambda or the existing json_escape in inference_client — 
        // since it's a static function there, duplicate the 10-line helper in xtype.cpp
        // as a local static or inline lambda.
        auto esc = [](std::string_view s) {
            std::string out;
            for (char c : s) {
                if (c == '"')  { out += "\\\""; }
                else if (c == '\\') { out += "\\\\"; }
                else if (c == '\n') { out += "\\n"; }
                else { out += c; }
            }
            return out;
        };
        std::snprintf(entry, sizeof(entry),
            "  {\"typed\":\"%s\",\"ghost\":\"%s\",\"app\":\"%s\","
            "\"accepted\":%s,\"latency_ms\":%d,\"ts\":%" PRId64 "}",
            esc(ev.typed).c_str(),
            esc(ev.ghost).c_str(),
            esc(ev.app).c_str(),
            ev.accepted ? "true" : "false",
            ev.latency_ms,
            (int64_t)ev.ts);
        json += entry;
    }
    json += "\n]\n";

    if (FILE* f = std::fopen(tmp.c_str(), "w")) {
        std::fputs(json.c_str(), f);
        std::fclose(f);
        std::rename(tmp.c_str(), dst.c_str());
    }
}
```

```cpp
// Read metrics.json
QFile mf(QDir::homePath() + "/.local/share/xtype/metrics.json");
if (mf.open(QIODevice::ReadOnly)) {
    QByteArray data = mf.readAll();
    mf.close();
    // Minimal JSON extraction — avoid Qt JSON overhead for a tiny file.
    // Look for "latency_p50_ms":NNN
    QRegularExpression re(R"("latency_p50_ms"\s*:\s*(\d+))");
    auto m = re.match(QString::fromUtf8(data));
    if (m.hasMatch()) {
        int p50 = m.captured(1).toInt();
        if (p50 != _latencyP50) {
            _latencyP50 = p50;
            emit latencyP50Changed();
        }
    }
}
```

## Step E2.4 — Settings app: RecentEventsModel reads recent_events.json

`LiveDemo.qml` calls `RecentEvents.first()` to seed the demo typewriter.

**File:** `settings/qt6-app/src/recent_events_model.h`

Add a `refresh()` invokable and a `QFileSystemWatcher` member:

```cpp
class RecentEventsModel : public QObject {
    Q_OBJECT
public:
    explicit RecentEventsModel(QObject* parent = nullptr);

    Q_INVOKABLE QVariantMap first() const;    // newest event or {}
    Q_INVOKABLE void refresh();

signals:
    void eventsChanged();

private:
    void loadFile();

    QVariantList    _events;
    QFileSystemWatcher _watcher;
    QString         _path;
};
```

**File:** `settings/qt6-app/src/recent_events_model.cpp`

```cpp
RecentEventsModel::RecentEventsModel(QObject* parent)
    : QObject(parent),
      _path(QDir::homePath() + "/.local/share/xtype/recent_events.json")
{
    if (QFile::exists(_path))
        _watcher.addPath(_path);
    connect(&_watcher, &QFileSystemWatcher::fileChanged,
            this, &RecentEventsModel::loadFile);
    loadFile();
}

QVariantMap RecentEventsModel::first() const {
    return _events.isEmpty() ? QVariantMap{} : _events.first().toMap();
}

void RecentEventsModel::refresh() { loadFile(); }

void RecentEventsModel::loadFile() {
    QFile f(_path);
    if (!f.open(QIODevice::ReadOnly)) return;
    QJsonDocument doc = QJsonDocument::fromJson(f.readAll());
    f.close();
    if (!doc.isArray()) return;
    _events.clear();
    for (const auto& val : doc.array())
        if (val.isObject())
            _events.append(val.toObject().toVariantMap());
    if (!_watcher.files().contains(_path) && QFile::exists(_path))
        _watcher.addPath(_path);
    emit eventsChanged();
}
```

Add `#include <QJsonDocument>`, `#include <QJsonArray>`,
`#include <QFileSystemWatcher>`, `#include <QFile>`, `#include <QDir>`.
Link `Qt6::Core` is already present; no new CMake changes needed.

**`LiveDemo.qml`** already checks `RecentEvents.first()` and falls back
to fixture data when empty. After E2.4, it will show real events when
the engine has produced any. No QML changes needed.
